图的存储要兼顾"判边快慢"与"空间开销"两件事。两种方式各有侧重：**邻接矩阵**用二维数组、`$O(n^2)$` 空间换 `$O(1)$` 判边，适合**稠密图**；**邻接表**用 `$n$` 条单链表、`$O(n+e)$` 空间换遍历邻接点的便利，适合**稀疏图**。

| 存储结构 | 空间 | 判边 | 适合 |
|:---|:---|:---|:---|
| 邻接矩阵 | `$O(n^2)$` | `$O(1)$` | 稠密图 |
| 邻接表 | `$O(n+e)$` | `$O(\deg)$`（遍历链表） | 稀疏图 |

## 邻接矩阵

用二维数组 `adj[][]` 存储顶点间的边/弧关系，物理上是一张 `$n\times n$` 的表。

### 结构定义

```c
#define MAX_VEX 100              // 最大顶点数
#define INF 0x3f3f3f3f           // 表示无边的"无穷大"

typedef enum {                   // 图的种类标志
    DG,      // 有向图
    AG,      // 无向图
    WDG,     // 有向网（带权）
    WAG      // 无向网（带权）
} GraphKind;

typedef struct {
    GraphKind kind;              // 图的种类标志
    int vexnum, arcnum;          // 当前顶点数和弧/边数
    int vexs[MAX_VEX];           // 顶点向量（存储顶点数据）
    int adj[MAX_VEX][MAX_VEX];   // 邻接矩阵（存储边/弧信息）
} MGraph;
```

### 关键性质

| 图类型 | 元素值 | 特殊性质 |
|:---|:---|:---|
| 无向图 | 1 有边 / 0 无边 | **对称矩阵**；第 `$i$` 行非零数 = 度；1 的总数 = `$2e$` |
| 无向带权图 | `$w_{ij}$` / `$\infty$` | 对称矩阵；第 `$i$` 行非 `$\infty$` 数 = 度；非 `$\infty$` 总数 = `$2e$` |
| 有向带权图 | `$w_{ij}$` / `$\infty$` | 一般**不对称**；第 `$i$` 行非 `$\infty$` 数 = 出度；第 `$j$` 列非 `$\infty$` 数 = 入度 |

### 基本操作

**插入顶点**：

```c++
int InsertVex(MGraph &G, int v) {
    int k = G.vexnum;                  // 新顶点的下标
    G.vexs[G.vexnum++] = v;            // 存入顶点向量，vexnum 自增

    if (G.kind == DG || G.kind == AG) // 不带权：邻接矩阵新行列置 0
        for (int j = 0; j < G.vexnum; j++)
            G.adj[j][k] = G.adj[k][j] = 0;
    else                               // 带权：置 ∞
        for (int j = 0; j < G.vexnum; j++)
            G.adj[j][k] = G.adj[k][j] = INF;

    return k;                          // 返回新顶点的下标
}
```

**插入边/弧**：

```c++
typedef struct {
    int vex1, vex2;                        // 弧的两个端点
    int ArcVal;                            // 权值
} ArcType;

int InsertArc(MGraph *G, ArcType *arc) {
    int k = LocateVex(*G, arc->vex1);      // 定位弧尾下标
    int j = LocateVex(*G, arc->vex2);      // 定位弧头下标
    if (G->kind == DG || G->kind == WDG) {  // 有向图 / 有向网
        G->adj[k][j] = arc->ArcVal;
    } else {                                // 无向图 / 无向网：对称赋值
        G->adj[k][j] = arc->ArcVal;
        G->adj[j][k] = arc->ArcVal;
    }
    G->arcnum++;
    return 1;                               // 返回 1 表示插入成功
}
```

**求第一个/下一个邻接点**（遍历邻接点的标准接口）：

```c++
int FirstAdjVex(MGraph G, int v) {
    int i = LocateVex(G, v);
    for (int j = 0; j < G.vexnum; j++)
        if (G.adj[i][j] != 0 && G.adj[i][j] != INF)
            return j;                // 返回第一个有边/弧的邻接点
    return -1;                       // 没有邻接点
}

int NextAdjVex(MGraph G, int v, int w) {
    int i = LocateVex(G, v);
    int j = LocateVex(G, w);
    for (int k = j + 1; k < G.vexnum; k++)
        if (G.adj[i][k] != 0 && G.adj[i][k] != INF)
            return k;                // 返回 w 之后第一个有边的邻接点
    return -1;
}
```

**删除顶点**（压缩行、列）：

```c++
void DeleteVex(MGraph &G, int v) {
    int k = LocateVex(G, v);
    for (int i = k; i < G.vexnum - 1; i++)    // ① 压缩 vexs[]
        G.vexs[i] = G.vexs[i + 1];
    for (int i = k; i < G.vexnum - 1; i++)    // ② 上移行
        for (int j = 0; j < G.vexnum; j++)
            G.adj[i][j] = G.adj[i + 1][j];
    for (int j = k; j < G.vexnum - 1; j++)    // ③ 左移列
        for (int i = 0; i < G.vexnum; i++)
            G.adj[i][j] = G.adj[i][j + 1];
    G.vexnum--;
}
```

**删除边/弧**：

```c++
void DeleteArc(MGraph &G, int u, int v) {
    int i = LocateVex(G, u);
    int j = LocateVex(G, v);
    G.adj[i][j] = (G.kind >= 2) ? INF : 0;  // 有权图置 INF，无权置 0
    if (G.kind == AG || G.kind == WAG)
        G.adj[j][i] = (G.kind >= 2) ? INF : 0;
    G.arcnum--;
}
```

### 复杂度

| 操作 | 时间复杂度 |
|:---|:---|
| 插入顶点 | `$O(n)$`（初始化新行列） |
| 插入/删除边 | `$O(n)$`（LocateVex 遍历）/ `$O(1)$`（已知下标） |
| 求第一个/下一个邻接点 | `$O(n)$`（遍历一行） |
| 删除顶点 | `$O(n^2)$`（双重循环压缩行列） |

> **易错点**：邻接矩阵的"第 `$i$` 行非零数 = 度"只对**无向图**成立；有向图的行统计的是**出度**、列统计的是**入度**，两者不等，审题时务必分清。

## 邻接表

对图中每个顶点建立一个**单链表**，链表中结点存放该顶点的所有邻接点。`$n$` 个顶点就有 `$n$` 个链表。

### 结点结构

```
vertex table (VNode)           edge node (ArcNode)
┌────────┬──────────┐         ┌──────┬──────┬──────┐
│  data  │ firstarc │────────→│adjvex│weight│ next │──→ ...
└────────┴──────────┘         └──────┴──────┴──────┘
```

- `adjvex`：邻接点在顶点表中的**下标**（关键字段）
- `weight`：边的权值（无权图可省略或置 1）
- `next`：指向下一个边表结点（同一顶点的下一条出边）

**有向图 vs 无向图**：

- **无向图**：每条边 `$(u,v)$` 出现两次，边表结点总数 `$ = 2e$`
- **有向图**：每条弧只在弧尾的链表里出现一次，边表结点总数 `$ = e$`

### 结构定义

```c
#define MAX_VEX 100
#define INF 0x3f3f3f3f

typedef int InfoType;                     // 与边/弧相关的信息类型（如权值）
typedef int VexType;                      // 顶点数据类型

/* ---- 弧/边的结构定义 ---- */
typedef struct {
    int vex1, vex2;                       // 弧或边所依附的两个顶点
    InfoType info;                        // 与边或弧相关的信息，如权值
} ArcType;

/* ---- 邻接表的结点定义 ---- */
typedef struct LinkNode {
    int adjvex;                           // 邻接点在头结点数组中的位置（下标）
    InfoType info;                        // 与边或弧相关的信息，如权值
    struct LinkNode *nextarc;             // 指向下一个表结点
} LinkNode;                               // 表结点类型定义

/* --- 顶点结点的结点定义 --- */
typedef struct {                          // 顶点结点（头结点）
    VexType data;                         // 顶点数据
    int degree;                           // 顶点的度（无向图）或出度（有向图）
    LinkNode *firstarc;                   // 第一条出边的链头
} VexNode;

/* ----- 图的结构定义 ----- */
typedef struct {
    GraphKind kind;                       // 图的种类标志
    int vexnum;                           // 顶点数
    VexNode AdjList[MAX_VEX];             // 邻接表（顶点数组）
} ALGraph;
```

### 基本操作

**增加顶点**：

```c++
int AddVertex(ALGraph *G, VexType *vp) {
    int k;
    G->AdjList[G->vexnum].data = *vp;      // 在末尾存入顶点数据
    G->AdjList[G->vexnum].degree = 0;      // 初始入/出度为 0
    G->AdjList[G->vexnum].firstarc = NULL; // 边链表初始为空
    k = ++G->vexnum;                       // 顶点数 +1，记录新下标
    return k;                              // 返回新顶点的下标
}
```

**增加弧**（头插法，无向图插两条链）：

```c++
int AddArc(ALGraph *G, ArcType *arc) {
    int k, j;
    LinkNode *p, *q;
    k = LocateVex(G, &arc->vex1);            // 定位弧尾下标
    j = LocateVex(G, &arc->vex2);            // 定位弧头下标

    p = (LinkNode *)malloc(sizeof(LinkNode));
    p->adjvex = arc->vex1;                   // 弧尾顶点
    p->info = arc->info;
    p->nextarc = NULL;

    q = (LinkNode *)malloc(sizeof(LinkNode));
    q->adjvex = arc->vex2;                   // 弧头顶点
    q->info = arc->info;
    q->nextarc = NULL;

    if (G->kind == AG || G->kind == WAG) {   // 无向图：头插两个链表
        q->nextarc = G->AdjList[k].firstarc;
        G->AdjList[k].firstarc = q;          // 弧头插入弧尾顶点的边表
        p->nextarc = G->AdjList[j].firstarc;
        G->AdjList[j].firstarc = p;          // 弧尾插入弧头顶点的边表
    } else {                                  // 有向图：头插弧尾链表
        q->nextarc = G->AdjList[k].firstarc;
        G->AdjList[k].firstarc = q;          // 建立正邻接链表
    }
    return 1;                                 // 返回 1 表示插入成功
}
```

> **考点**：邻接表插入边用**头插法**，新结点始终插在链表头部，复杂度严格 `$O(1)$`，与链表长度无关。

## 邻接矩阵 vs 邻接表

| 维度 | 邻接矩阵 | 邻接表 |
|:---|:---|:---|
| 空间 | `$O(n^2)$` | `$O(n+e)$` |
| 判边 | `$O(1)$` | `$O(\deg)$`（需遍历链表） |
| 遍历邻接点 | `$O(n)$` | `$O(\deg_i)$` |
| 入度（有向图） | 遍历列 `$O(n)$` | 需逆邻接表 |
| 适合 | 稠密图 | 稀疏图 |

> **易错点**：**遍历结果唯一性不同**——邻接矩阵（基于顶点下标顺序）的遍历结果**唯一**；邻接表依赖链表的内存/插入顺序，遍历结果**不唯一**。

## 引子：十字链表与邻接多重表

当两种基础结构各自有短板时，另有两种"升级版"链式存储专门补齐：

- **十字链表（有向图）**：每个弧结点同时挂出弧链（`tlink`）与入弧链（`hlink`），可**同时高效求得出度与入度**。
- **邻接多重表（无向图）**：每条边只存**一次**，删除边/弧、判断两顶点是否邻接都更方便，避免了邻接表每条无向边存两次的冗余。

## 高频考点速记

| 考点 | 记忆 |
|:---|:---|
| 邻接矩阵 | 空间 `$O(n^2)$`、判边 `$O(1)$，适合**稠密图** |
| 无向图矩阵 | 对称；第 `$i$` 行非零数 = 度；1 总数 = `$2e$` |
| 有向图矩阵 | 第 `$i$` 行非零数 = **出度**、第 `$j$` 列非零数 = **入度** |
| 邻接表 | 空间 `$O(n+e)$`、判边 `$O(\deg)$，适合**稀疏图** |
| 边表结点数 | 无向图边表结点 `$2e$` 个、有向图 `$e$` 个 |
| 头插法 | 新结点始终插入链表头部 `$O(1)`，不依赖链表长度 |
| 遍历唯一性 | 邻接矩阵结果**唯一**；邻接表**不唯一**（依赖链表顺序） |
| 升级结构 | 有向图求入度用十字链表、无向图删边用邻接多重表 |
